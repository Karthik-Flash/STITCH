```verilog
module register_4bit (
    input [3:0] din,
    input clk, clr, ce,
    output [3:0] dout
);

reg [3:0] dout_reg;

always @(posedge clk) begin
    if (clr == 1'b1) begin
        dout_reg <= 4'b0000;
    end else if (ce == 1'b1) begin
        dout_reg <= din;
    end
end

assign dout = dout_reg;

endmodule
```
