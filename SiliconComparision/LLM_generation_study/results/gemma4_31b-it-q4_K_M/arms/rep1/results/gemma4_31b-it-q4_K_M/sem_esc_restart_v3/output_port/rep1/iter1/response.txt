```verilog
module output_port (
    input clk,
    input reset,
    input [3:0] req,
    input [31:0] flit0,
    input [31:0] flit1,
    input [31:0] flit2,
    input [31:0] flit3,
    input full_in,
    output [31:0] data_out,
    output [3:0] pop
);

    reg [2:0] owner; // 0-3: requester, 4: free
    reg [31:0] data_out_reg;
    reg [3:0] selected;
    wire [3:0] gnt;
    wire [31:0] mux_out;
    wire [3:0] pop_wire;
    wire advance;

    assign data_out = data_out_reg;
    assign pop = pop_wire;

    // Selected requester logic
    always @(*) begin
        if (owner == 3'd4) begin
            selected = gnt;
        end else begin
            selected = 4'b0000;
            selected[owner] = 1'b1;
        end
    end

    // Pop signal: one-hot, indicates which flit will be sent.
    // Must be 0 during reset as "Nothing is sent at that edge".
    assign pop_wire = (reset) ? 4'b0000 : ((selected & req) & (~full_in));

    // Advance signal for round-robin arbiter: 
    // "if the output was free before the edge, the selected requester becomes the last granted requester."
    assign advance = (owner == 3'd4) && (pop_wire != 4'b0000);

    // Instantiate round-robin arbiter
    rr_arbiter arb_inst (
        .clk(clk),
        .reset(reset),
        .req(req),
        .advance(advance),
        .gnt(gnt)
    );

    // Instantiate flit multiplexer
    flit_mux mux_inst (
        .sel(pop_wire),
        .in0(flit0),
        .in1(flit1),
        .in2(flit2),
        .in3(flit3),
        .out(mux_out)
    );

    // State update logic
    always @(posedge clk) begin
        if (reset) begin
            owner <= 3'd4;
            data_out_reg <= 32'h6000_0000;
        end else begin
            if (pop_wire != 4'b0000) begin
                // A flit is sent
                data_out_reg <= mux_out;
                if (mux_out[30:29] == 2'b10) begin
                    // Tail flit: output becomes free
                    owner <= 3'd4;
                end else begin
                    // Head or Body flit: requester becomes/stays owner
                    if (pop_wire[0]) begin
                        owner <= 3'd0;
                    end else if (pop_wire[1]) begin
                        owner <= 3'd1;
                    end else if (pop_wire[2]) begin
                        owner <= 3'd2;
                    end else if (pop_wire[3]) begin
                        owner <= 3'd3;
                    end else begin
                        owner <= 3'd4;
                    end
                end
            end else begin
                // No flit is sent
                data_out_reg <= 32'h6000_0000;
                owner <= owner;
            end
        end
    end

endmodule
```
