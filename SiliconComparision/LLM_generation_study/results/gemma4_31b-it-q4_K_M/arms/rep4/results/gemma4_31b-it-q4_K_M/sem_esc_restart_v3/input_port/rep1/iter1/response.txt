```verilog
module input_port (
    input clk,
    input reset,
    input [31:0] data_in,
    input [7:0] current_address,
    input rd_en,
    output full_out,
    output [31:0] flit,
    output [4:0] req
);

    // Internal signals
    wire [31:0] fifo_dout;
    wire fifo_empty;
    wire fifo_almost_full;
    wire [4:0] computed_req;
    reg [4:0] held_route;
    reg [4:0] req_out;
    wire wr_en_fifo;
    wire rd_en_fifo;

    // FIFO control signals
    // data_in is stored if type is not 11 (idle) and reset is not active.
    // Flit type is bits [30:29].
    assign wr_en_fifo = (!reset && (data_in[30:29] != 2'b11));
    // rd_en is ignored during reset.
    assign rd_en_fifo = (!reset && rd_en);

    // Instantiate FIFO buffer
    fifo u_fifo (
        .clk(clk),
        .reset(reset),
        .wr_en(wr_en_fifo),
        .din(data_in),
        .rd_en(rd_en_fifo),
        .dout(fifo_dout),
        .empty(fifo_empty),
        .almost_full(fifo_almost_full)
    );

    // Instantiate Route Compute logic
    route_compute u_route_compute (
        .flit(fifo_dout),
        .current_address(current_address),
        .req(computed_req)
    );

    // Held route register: updated when a head flit is popped from the buffer.
    // Flit type 01 = head.
    always @(posedge clk) begin
        if (reset) begin
            held_route <= 5'b0;
        end else if (rd_en && !fifo_empty && (fifo_dout[30:29] == 2'b01)) begin
            held_route <= computed_req;
        end else begin
            held_route <= held_route;
        end
    end

    // Combinational logic for the route request (req)
    always @(*) begin
        if (fifo_empty) begin
            req_out = 5'b0;
        end else if (fifo_dout[30:29] == 2'b01) begin
            // Front flit is a head flit: use computed route
            req_out = computed_req;
        end else begin
            // Front flit is body or tail: use held route
            req_out = held_route;
        end
    end

    // Output assignments
    assign full_out = fifo_almost_full;
    assign flit = fifo_dout;
    assign req = req_out;

endmodule
```
