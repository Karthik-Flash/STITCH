```verilog
module input_port (
    input clk,
    input reset,
    input [31:0] data_in,
    input [7:0] current_address,
    input rd_en,
    output full_out,
    output [31:0] flit,
    output reg [4:0] req
);

    wire [31:0] fifo_dout;
    wire fifo_empty;
    wire fifo_almost_full;
    wire [4:0] computed_req;
    reg [4:0] held_route;
    wire wr_en;

    // Buffer write enable: store if not idle and not resetting
    assign wr_en = (!reset) && (data_in[30:29] != 2'b11);

    // FIFO instantiation
    fifo u_fifo (
        .clk(clk),
        .reset(reset),
        .wr_en(wr_en),
        .din(data_in),
        .rd_en(rd_en),
        .dout(fifo_dout),
        .empty(fifo_empty),
        .almost_full(fifo_almost_full)
    );

    // Route computation instantiation
    route_compute u_route_compute (
        .flit(fifo_dout),
        .current_address(current_address),
        .req(computed_req)
    );

    // Held route update logic
    always @(posedge clk) begin
        if (reset) begin
            held_route <= 5'b0;
        end else if (rd_en && !fifo_empty && (fifo_dout[30:29] == 2'b01)) begin
            held_route <= computed_req;
        end
        // else held_route <= held_route; // Implicitly held
    end

    // Request output logic
    always @(*) begin
        if (fifo_empty) begin
            req = 5'b0;
        end else if (fifo_dout[30:29] == 2'b01) begin
            req = computed_req;
        end else if (fifo_dout[30:29] == 2'b00 || fifo_dout[30:29] == 2'b10) begin
            req = held_route;
        end else begin
            req = 5'b0;
        end
    end

    // Output assignments
    assign flit = fifo_dout;
    assign full_out = fifo_almost_full;

endmodule
```
