```verilog
module flit_mux (
    input [3:0] sel,
    input [31:0] in0,
    input [31:0] in1,
    input [31:0] in2,
    input [31:0] in3,
    output [31:0] out
);

    reg [31:0] out_reg;

    always @(*) begin
        case (sel)
            4'b0001: begin
                out_reg = in0;
            end
            4'b0010: begin
                out_reg = in1;
            end
            4'b0100: begin
                out_reg = in2;
            end
            4'b1000: begin
                out_reg = in3;
            end
            default: begin
                out_reg = 32'h6000_0000;
            end
        endcase
    end

    assign out = out_reg;

endmodule
```
