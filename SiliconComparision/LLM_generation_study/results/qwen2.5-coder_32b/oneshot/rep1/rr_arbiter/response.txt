```verilog
module rr_arbiter (
    input clk,
    input reset,
    input [3:0] req,
    input advance,
    output [3:0] gnt
);

    reg [1:0] last;
    wire [3:0] gnt_comb;

    always @(posedge clk) begin
        if (reset) begin
            last <= 2'b11; // Last requester becomes 3
        end else if (advance && |gnt_comb) begin
            last <= gnt_comb[0] ? 2'b00 :
                   gnt_comb[1] ? 2'b01 :
                   gnt_comb[2] ? 2'b10 : 2'b11;
        end
    end

    always @(*) begin
        case (last)
            2'b00: begin
                if (req[1]) gnt_comb = 4'b0010;
                else if (req[2]) gnt_comb = 4'b0100;
                else if (req[3]) gnt_comb = 4'b1000;
                else if (req[0]) gnt_comb = 4'b0001;
                else gnt_comb = 4'b0000;
            end
            2'b01: begin
                if (req[2]) gnt_comb = 4'b0100;
                else if (req[3]) gnt_comb = 4'b1000;
                else if (req[0]) gnt_comb = 4'b0001;
                else if (req[1]) gnt_comb = 4'b0010;
                else gnt_comb = 4'b0000;
            end
            2'b10: begin
                if (req[3]) gnt_comb = 4'b1000;
                else if (req[0]) gnt_comb = 4'b0001;
                else if (req[1]) gnt_comb = 4'b0010;
                else if (req[2]) gnt_comb = 4'b0100;
                else gnt_comb = 4'b0000;
            end
            2'b11: begin
                if (req[0]) gnt_comb = 4'b0001;
                else if (req[1]) gnt_comb = 4'b0010;
                else if (req[2]) gnt_comb = 4'b0100;
                else if (req[3]) gnt_comb = 4'b1000;
                else gnt_comb = 4'b0000;
            end
            default: gnt_comb = 4'b0000;
        endcase
    end

    assign gnt = gnt_comb;

endmodule
```
