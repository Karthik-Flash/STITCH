```verilog
module fifo (
    input clk,
    input reset,
    input wr_en,
    input [31:0] din,
    input rd_en,
    output [31:0] dout,
    output empty,
    output almost_full
);

    reg [31:0] mem [3:0];
    reg [1:0] wr_ptr;
    reg [1:0] rd_ptr;
    reg [2:0] count;

    wire full = (count == 3'b100);
    wire almost_full_w = (count >= 3'b10);

    assign dout = mem[rd_ptr];
    assign empty = (count == 3'b00);
    assign almost_full = almost_full_w;

    always @(posedge clk) begin
        if (reset) begin
            wr_ptr <= 2'b00;
            rd_ptr <= 2'b00;
            count <= 3'b000;
        end else begin
            if (wr_en && !full && !rd_en) begin
                mem[wr_ptr] <= din;
                wr_ptr <= wr_ptr + 1;
                count <= count + 1;
            end else if (!wr_en && rd_en && !empty) begin
                rd_ptr <= rd_ptr + 1;
                count <= count - 1;
            end else if (wr_en && !full && rd_en && !empty) begin
                mem[wr_ptr] <= din;
                wr_ptr <= wr_ptr + 1;
                rd_ptr <= rd_ptr + 1;
            end
        end
    end

endmodule
```
