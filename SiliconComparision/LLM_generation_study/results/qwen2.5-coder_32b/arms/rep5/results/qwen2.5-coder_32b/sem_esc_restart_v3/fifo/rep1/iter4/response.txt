```verilog
module fifo (
    input clk,
    input reset,
    input wr_en,
    input [31:0] din,
    input rd_en,
    output [31:0] dout,
    output empty,
    output almost_full
);

    reg [31:0] mem [3:0];
    reg [1:0] wr_ptr;
    reg [1:0] rd_ptr;
    reg [2:0] count;

    wire full;
    wire not_empty;

    assign full = (count == 3'b100);
    assign not_empty = (count != 3'b000);
    assign dout = not_empty ? mem[rd_ptr] : 32'b0;
    assign empty = ~not_empty;
    assign almost_full = (count >= 3'b11);

    always @(posedge clk) begin
        if (reset) begin
            wr_ptr <= 2'b00;
            rd_ptr <= 2'b00;
            count <= 3'b000;
        end else begin
            case ({wr_en, rd_en, full, not_empty})
                4'b1111: begin
                    // Write and read at the same time, do nothing
                end
                4'b1101: begin
                    // Write when full, do nothing
                end
                4'b1100: begin
                    // Write when not full
                    mem[wr_ptr] <= din;
                    wr_ptr <= wr_ptr + 1;
                    count <= count + 1;
                end
                4'b0111: begin
                    // Read when not empty
                    rd_ptr <= rd_ptr + 1;
                    count <= count - 1;
                end
                4'b0110: begin
                    // Read when empty, do nothing
                end
                default: begin
                    // Default case, do nothing
                end
            endcase
        end
    end

endmodule
```
